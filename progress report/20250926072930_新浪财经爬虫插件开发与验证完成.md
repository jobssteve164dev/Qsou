```python
class SinaFinanceSpider(scrapy.Spider):
    name = "sina_finance"
    allowed_domains = ["finance.sina.com.cn", "sina.com.cn"]
    
    start_urls = [
        "https://finance.sina.com.cn/roll/index.d.html?cid=56588&page=1",
        "https://finance.sina.com.cn/stock/",
        "https://finance.sina.com.cn/money/fund/",
        "https://finance.sina.com.cn/money/bond/",
        "https://finance.sina.com.cn/china/",
    ]
    
    def parse_article(self, response):
        """解析新闻详情页面"""
        item = NewsArticleItem()
        item["title"] = response.css("h1::text").get() or ""
        
        # 优先使用trafilatura提取正文
        extracted = trafilatura.extract(response.text, include_comments=False, include_tables=False)
        if extracted:
            item["content"] = extracted.strip()
        else:
            # 回退到CSS选择器
            content_parts = response.css("div.article p::text, .content p::text").getall()
            item["content"] = "\n".join([p.strip() for p in content_parts if p and p.strip()])
        
        # 数据质量验证
        if self.validate_news_item(item):
            yield item
```
