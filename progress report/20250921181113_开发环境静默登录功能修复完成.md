```python
# api-gateway/app/api/v1/endpoints/auth.py
@router.post("/login", response_model=LoginResponse)
async def login(payload: LoginRequest):
    if not _is_auth_enabled():
        raise HTTPException(status_code=404, detail="Auth disabled in this environment")

    # 最小校验：接受演示账号 admin/admin123 与 user/user123
    valid = (
        (payload.username == "admin" and payload.password == "admin123")
        or (payload.username == "user" and payload.password == "user123")
    )
    if not valid:
        raise HTTPException(status_code=401, detail="Invalid credentials")

    return LoginResponse(token=_fake_token(payload.username), user=_fake_user(payload.username))
```

**示例: 前端路由守卫静默登录增强**
```typescript
// web-frontend/src/components/auth/AuthContext.tsx
const handleRouteChange = async (url: string) => {
  if (isProtectedRoute && !isAuthenticated) {
    // 在开发环境下，尝试静默登录一次，避免点击受保护页面立即被重定向
    const token = typeof window !== 'undefined' ? localStorage.getItem('auth_token') : null;
    if (!token && enableDevSilentLogin) {
      setLoading(true);
      const ok = await devSilentLogin();
      setLoading(false);
      if (ok) {
        // 已登录，继续原目标路由
        router.replace(url);
        return;
      }
    }
    router.push(`/login?returnUrl=${encodeURIComponent(url)}`);
  }
};
```

## 5. 自我评估与学习 (Self-Assessment & Learning)

*   **遇到的挑战**: 
    - 初始问题定位困难：需要同时分析前端路由守卫逻辑和后端API端点缺失
    - 前端构建错误：JSX语法在.ts文件中导致Next.js编译失败
    - 前后端协同调试：需要确保API接口格式与前端期望一致

*   **学到的教训**: 
    - 开发环境静默登录需要前后端完整配合，不能仅依赖前端逻辑
    - 文件扩展名对TypeScript/JSX编译的重要性，.tsx文件才能正确解析JSX语法
    - 路由守卫的时序问题：需要在重定向前给静默登录留出执行时间
    - 环境变量控制的重要性：开发功能应该有明确的开关机制
